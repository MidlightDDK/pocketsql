CREATE TABLE singer_in_concert (concert_ID BIGINT, Singer_ID BIGINT);
CREATE TABLE stadium (Stadium_ID BIGINT, Location VARCHAR /* e.g. 'Alloa Athletic', 'Arbroath' */, Name VARCHAR /* e.g. 'Balmoor', 'Bayview Stadium' */, Capacity BIGINT, Highest BIGINT, Lowest BIGINT, Average BIGINT);
